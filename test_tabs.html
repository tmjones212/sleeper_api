Testing Tab Functionality

Open the main index.html in a browser, then open developer console (F12) and run this test:

// Copy and paste this into the console:
function testTabs() {
    console.log('=== TESTING TAB FUNCTIONALITY ===');
    
    // Test if matchups content is visible
    showPanel('matchups');
    const mc = document.getElementById('matchupsContainer');
    const yearDivs = document.querySelectorAll('.year-matchups');
    
    console.log('matchupsContainer exists:', \!\!mc);
    console.log('matchupsContainer visible:', window.getComputedStyle(mc).display);
    console.log('Year divs found:', yearDivs.length);
    
    yearDivs.forEach(div => {
        const year = div.getAttribute('data-year');
        const display = window.getComputedStyle(div).display;
        const hasContent = div.querySelector('.matchup-card') \!== null;
        console.log(`Year ${year}: display=${display}, has content=${hasContent}`);
    });
    
    // Force show 2024
    window.showMatchupYear();
}

testTabs();
